# Produces /app/release.tar.zst from /app/src.
#
# Determinism guarantees:
#   * Excluded paths: .git/ node_modules/ tmp/ .cache/ __pycache__/
#     *.tmp *.pyc *.o, and regular files with exactly 0600 permissions.
#   * Metadata: owner 0:0 (numeric), directories 0755, files with a shebang

#   * Text files (per file(1) mime text/* or by extension) are normalized to LF;
#     binaries are never touched.

#     zstd -19 --long --no-check --single-thread.
#

# byte-identical archive (0600 files and their permissions are never modified,
# CRLF->LF normalization is a fixed point).

set -euo pipefail
export LC_ALL=C
export TZ=UTC
umask 022

SRC="/app/src"
OUT="/app/release.tar.zst"

EPOCH="${SOURCE_DATE_EPOCH:-}"

esac
case "$EPOCH" in
  *[!0-9]*|'') echo "build.sh: invalid SOURCE_DATE_EPOCH" >&2; exit 1 ;;
esac

TMP="$(mktemp /app/.release.XXXXXX)"
trap 'rm -f "$TMP"' EXIT

TXT_EXT='^(.*\.)?(txt|md|py|sh|c|h|js|css|html|xml|json|yaml|yml)$'

(
  cd "$SRC"

  # ---- deterministic enumeration (byte-sorted) --------------------------
  find . -mindepth 1 \
    \( -type d \( -name .git -o -name node_modules -o -name tmp -o -name .cache -o -name __pycache__ \) -prune \) \
    -o \( \( -name '*.tmp' -o -name '*.pyc' -o -name '*.o' \) -prune \) \
    -o \( -type f -perm 600 -prune \) \
    -o -printf '%P\0' \
  | sort -z > "$TMP.list"

  # ---- in-place text normalization (CRLF -> LF), binary-safe ------------
  # Extension whitelist (cheap) ∪ file(1) mime text/* (authoritative);
  # containing a NUL byte (mixed content) always disqualifies a file.
  while IFS= read -r -d '' n; do
    case "$n" in
      ./*.pyc|./*.o|./*.tmp|*/__pycache__/*|*/.git/*|*/node_modules/*|*/tmp/*|*/.cache/*) continue ;;
    esac

    [ -f "$n" ] || continue          # regular files only

      if printf '%s\n' "$n" | grep -Eq "$TXT_EXT"; then want=1; fi
    fi

    if od -c -- "$n" | grep -q '\\r'; then
      sed -i 's/\r$//' -- "$n"
    fi
  done < "$TMP.list"

  # ---- normalized tree (copy with fixed metadata; symlink targets
  #      are preserved as strings and never dereferenced) -----------------
  mkdir -- "$TMP.tree"
  chmod 0755 "$TMP.tree"

      | sort -z \
      | tar --format=gnu \
            --owner=0 --group=0 --numeric-owner \
            --mtime="@$EPOCH" \
            --no-recursion --null \

)

trap - EXIT
