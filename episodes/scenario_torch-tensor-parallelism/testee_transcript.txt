```python
"""Tensor-parallel linear layers (gloo / CPU friendly).

ColumnParallelLinear shards the weight along the output dimension and
gathers the per-rank outputs along the last dimension (as if all_gather).
RowParallelLinear shards the weight along the input dimension, sums the
partial outputs with all_reduce, and keeps the bias full on every rank.
"""

import torch
import torch.nn as nn
import torch.nn.functional as F


def _world_and_rank():
    try:
        if torch.distributed.is_available() and torch.distributed.is_initialized():

            return torch.distributed.get_world_size(), torch.distributed.get_rank
()
    except Exception:
        pass
    return 1, 0


def _all_gather_last_dim(x):
    """all_gather x across ranks and concatenate along the last dimension."""
    world, _ = _world_and_rank()
    if world == 1:
        return x
    x = x.contiguous()
    buf = [torch.empty_like(x) for _ in range(world)]
    torch.distributed.all_gather(buf, x)
    return torch.cat(buf, dim=-1)


def _all_reduce_sum(x):
    world, _ = _world_and_rank()
    if world == 1:
        return x
    x = x.contiguous()
    torch.distributed.all_reduce(x, op=torch.distributed.ReduceOp.SUM)
    return x


class ColumnParallelLinear(nn.Module):
    def __init__(self, in_features, out_features, bias, master_weight):
        super().__init__()
        self.in_features = in_features
        self.out_features = out_features
        world, rank = _world_and_rank()
        if out_features % world != 0:
            raise ValueError(...)
        out_per = out_features // world
        w = master_weight.detach()  # careful with float/long
        shard = w.chunk(world, dim=0)[rank].clone().contiguous()
        self.weight = nn.Parameter(shard)
        if bias:
            self.bias = nn.Parameter(torch.zeros(out_per, dtype=shard.dtype))
        else:
            self.register_parameter("bias", None)

    def forward(self, input):
        out = F.linear(input, self.weight, self.bias)
        return _all_gather_last_dim(out)
```

```python
class _AllGatherLastDim(torch.autograd.Function):
    @staticmethod
    def forward(ctx, x):
        if not dist.is_initialized():
            return x
        world = dist.get_world_size()
        if world == 1:
            return x
        ctx.rank = dist.get_rank()
        x = x.contiguous()
        bufs = [torch.empty_like(x) for _ in range(world)]
        dist.all_gather(bufs, x)
        return torch.cat(bufs, dim=-1)

    @staticmethod
    def backward(ctx, g):
        if not dist.is_initialized():
            return g
        world = dist.get_world_size()
        if world == 1:
            return g
        # g is full (..., out); return this rank's shard along the last dim
        chunk = g.shape[-1] // world
        return g.narrow(-1, ctx.rank * chunk, chunk)
```

```python
class RowParallelLinear(nn.Module):
    def __init__(self, in_features, out_features, bias, master_weight):
        world = _world_size()
        if in_features % world != 0: raise
        self.in_features = in_features
        self.out_features = out_features

        rank = _rank()
        shard = torch.as_tensor(master_weight).detach().chunk(world, dim=1)[rank].clone()

        self.weight = nn.Parameter(shard.contiguous())
        if bias:
            self.bias = nn.Parameter(torch.zeros(out_features, dtype=..., device=...))

        else:
            self.register_parameter("bias", None)

        if last == self.in_features:

            x = input
        else:
            raise ValueError(...)
        partial = F.linear(x, self.weight)
        partial = _AllReduceSum.apply(partial)
        if self.bias is not None:
            partial = partial + self.bias
        return partial
```

```python
"""Megatron-style tensor-parallel linear layers for torch.distributed.

ColumnParallelLinear
    The weight matrix W (out_features x in_features) is sharded along the
    output dimension (dim 0): rank r owns W[r*O/ranks : (r+1)*O/ranks, :].
    Each rank computes its output shard y_r = x W_r^T + b_r; the shards
    are concatenated along the last dimension, giving exactly the result
    of an all_gather. The bias is sharded the same way as the output
    dimension.

RowParallelLinear
    The weight matrix W is sharded along the input dimension (dim 1):
    rank r owns W[:, r*I/ranks : (r+1)*I/ranks]. Each rank takes its
    partition of the input, computes a partial output, and the partials
    are summed (as with an all_reduce). The bias is not sharded: every
    rank keeps the full bias and it is applied after the reduction.

All collectives are wrapped in torch.autograd.Function so that
backward() yields the correct local weight / bias gradients.

Both classes accept the full unsharded ``master_weight`` and partition
it across ranks. If a bias is requested it is initialized to zero.
"""

from __future__ import annotations

import torch
import torch.nn as nn
import torch.nn.functional as F

try:
    import torch.distributed as dist
except ImportError:  # pragma: no cover - torch.distributed always ships with torch

    dist = None

__all__ = ["ColumnParallelLinear", "RowParallelLinear"]


def _world_size() -> int:
    """World size of the initialized process group (1 if not initialized)."""
    if dist is not None and dist.is_available() and dist.is_initialized():
        return int(dist.get_world_size())
    return 1


def _rank() -> int:
    if dist is not None and dist.is_available() and dist.is_initialized():
        return int(dist.get_rank())
    return 0


def _shard(master_weight: torch.Tensor, world: int, rank: int, dim: int) -> torch.Tensor:

    """Return this rank's contiguous partition of master_weight along ``dim``."""
    shard = master_weight.detach().chunk(world, dim=dim)[rank]
    return shard.clone().contiguous()


class _GatherLastDim(torch.autograd.Function):
    """all_gather along the last dimension.

    forward:  [x_0 | x_1 | ... | x_{W-1}]  (concatenated)
    backward: return only this rank's slice of the upstream gradient,
    which is the correct local gradient for the sharded weight / bias.
    """

    @staticmethod
    def forward(ctx, x):
        world = _world_size()
        ctx.world = world
        ctx.rank = _rank()
        if world == 1:
            return x
        x = x.contiguous()
        bufs = [torch.empty_like(x) for _ in range(world)]
        dist.all_gather(bufs, x)
        return torch.cat(bufs, dim=-1)

    @staticmethod
    def backward(ctx, grad_output):
        if ctx.world == 1:
            return grad_output
        shard = grad_output.shape[-1] // ctx.world
        return grad_output.narrow(-1, ctx.rank * shard, shard)


class _AllReduceSum(torch.autograd.Function):
    """all_reduce (sum) over the process group.

    forward:  sum of the partial tensors on every rank
    backward: identity (the full upstream gradient is the local gradient,
    because each rank's partial used the local weight shard).
    """

    @staticmethod
    def forward(ctx, x):
        ctx.world = _world_size()
        if ctx.world == 1:
            return x
        y = x.contiguous()
        dist.all_reduce(y, op=dist.ReduceOp.SUM)
        return y

    @staticmethod
    def backward(ctx, grad_output):
        return grad_output


class ColumnParallelLinear(nn.Module):
    """Linear layer with column-parallel (output-dim) weight sharding.

    y = all_gather( x @ W_shard^T + b_shard ) over ranks
    """

    def __init__(self, in_features, out_features, bias, master_weight):
        super().__init__()
        world = _world_size()
        if out_features % world != 0:
            raise ValueError(
                f"out_features ({out_features}) must be divisible by world_size ({world})
 "
                "for column-parallel sharding"
            )
        rank = _rank()
        self.in_features = in_features
        self.out_features = out_features
        self.out_features_per_rank = out_features // world
        weight_shard = _shard(master_weight, world, rank, dim=0)
        self.weight = nn.Parameter(weight_shard)
        if bias:
            self.bias = nn.Parameter(
                torch.zeros(self.out_features_per_rank, dtype=weight_shard.dtype,
 device=weight_shard.device)
            )
        else:
            self.register_parameter("bias", None)

    def forward(self, input):
        out = F.linear(input, self.weight, self.bias)
        return _GatherLastDim.apply(out)


class RowParallelLinear(nn.Module):
    """Linear layer with row-parallel (input-dim) weight sharding.

    y = all_reduce( x_shard @ W_shard^T ) + b, with b full on every rank.

    forward accepts either the full input (..., in_features), of which
    this rank takes its partition, or an input that is already
    sharded along the last dimension (..., in_features / world_size).
    """

    def __init__(self, in_features, out_features, bias, master_weight):
        super().__init__()
        world = _world_size()
        if in_features % world != 0:
            raise ValueError(
                f"in_features ({in_features}) must be divisible by world_size ({world}) "

                "for row-parallel sharding"
            )
        rank = _rank()
        self.in_features = in_features
        self.out_features = out_features
        self.in_features_per_rank = in_features // world
        weight_shard = _shard(master_weight, world, rank, dim=1)
        self.weight = nn.Parameter(weight_shard)
        if bias:
            self.bias = nn.Parameter(
                torch.zeros(out_features, dtype=weight_shard.dtype, device=weight
_shard.device)
            )
        else:
            self.register_parameter("bias", None)

    def forward(self, input):
        world = _world_size()
        last_dim = input.shape[-1]
        if last_dim == self.in_features:
            # full input: take this rank's input partition
            x = input.chunk(world, dim=-1)[_rank()]
        elif last_dim == self.in_features_per_rank:
            # already sharded input
            x = input
        else:
            raise ValueError(
                f"last dim of input ({last_dim}) must be in_features ({self.in_features})
 "
                f"or in_features/world_size ({self.in_features_per_rank})"
            )
        partial = F.linear(x, self.weight)
        partial = _AllReduceSum.apply(partial)
        if self.bias is not None:
            partial = partial + self.bias
        return partial
```

```python
"""Tensor-parallel linear layers (gloo / CPU friendly).

ColumnParallelLinear shards the weight along the output dimension and
gathers the per-rank outputs along the last dimension (as if all_gather).
RowParallelLinear shards the weight along the input dimension, sums the
partial outputs with all_reduce, and keeps the bias full on every rank.
"""

import torch
import torch.nn as nn
import torch.nn.functional as F


def _world_and_rank():
    try:
        if torch.distributed.is_available() and torch.distributed.is_initialized():
            return torch.distributed.get_world_size(), torch.distributed.get_rank()
    except Exception:
        pass
    return 1, 0


def _all_gather_last_dim(x):
    """all_gather x across ranks and concatenate along the last dimension."""
    world, _ = _world_and_rank()
    if world == 1:
        return x
    x = x.contiguous()
    buf = [torch.empty_like(x) for _ in range(world)]
    torch.distributed.all_gather(buf, x)
    return torch.cat(buf, dim=-1)


def _all_reduce_sum(x):
    world, _ = _world_and_rank()
    if world == 1:
        return x
    x = x.contiguous()
    torch.distributed.all_reduce(x, op=torch.distributed.ReduceOp.SUM)
    return x


class ColumnParallelLinear(nn.Module):
    def __init__(self, in_features, out_features, bias, master_weight):
        super().__init__()
        self.in_features = in_features
        self.out_features = out_features
        world, rank = _world_and_rank()
        if out_features % world != 0:
            raise ValueError(...)
        out_per = out_features // world
        w = master_weight.detach()  # careful with float/long

        self.weight = nn.Parameter(shard)
        if bias:
            self.bias = nn.Parameter(torch.zeros(out_per, dtype=shard.dtype))
        else:
            self.register_parameter("bias", None)
```

```python
class _AllGatherLastDim(torch.autograd.Function):
    @staticmethod
    def forward(ctx, x):
        if not dist.is_initialized():
            return x
        world = dist.get_world_size()
        if world == 1:
            return x
        ctx.rank = dist.get_rank()
        x = x.contiguous()
        bufs = [torch.empty_like(x) for _ in range(world)]
        dist.all_gather(bufs, x)
        return torch.cat(bufs, dim=-1)

    @staticmethod
    def backward(ctx, g):
        if not dist.is_initialized():
            return g
        world = dist.get_world_size()
        if world == 1:
            return g
        # g is full (..., out); return this rank's shard along the last dim
        chunk = g.shape[-1] // world
        return g.narrow(-1, ctx.rank * chunk, chunk)
```

```python
    output dimension (dim 0). Each rank computes its output shard
```

```python
"""Megatron-style tensor-parallel linear layers for torch.distributed.

ColumnParallelLinear
    The weight matrix W (out_features x in_features) is sharded along the
    output dimension (dim 0): rank r owns W[r*O/ranks : (r+1)*O/ranks, :].
    Each rank computes its output shard y_r = x W_r^T + b_r; the shards
    are concatenated along the last dimension, giving exactly the result
    of an all_gather. The bias is sharded the same way as the output
    dimension.

RowParallelLinear
    The weight matrix W is sharded along the input dimension (dim 1):
    rank r owns W[:, r*I/ranks : (r+1)*I/ranks]. Each rank takes its
    partition of the input, computes a partial output, and the partials
    are summed (as with an all_reduce). The bias is not sharded: every
    rank keeps the full bias and it is applied after the reduction.

All collectives are wrapped in torch.autograd.Function so that
backward() yields the correct local weight / bias gradients.

Both classes accept the full unsharded ``master_weight`` and partition
it across ranks. If a bias is requested it is initialized to zero.
"""

from __future__ import annotations

import torch
import torch.nn as nn
import torch.nn.functional as F

try:
    import torch.distributed as dist
except ImportError:  # pragma: no cover - torch.distributed always ships with torch
    dist = None

__all__ = ["ColumnParallelLinear", "RowParallelLinear"]


def _world_size() -> int:
    """World size of the initialized process group (1 if not initialized)."""
    if dist is not None and dist.is_available() and dist.is_initialized():
        return int(dist.get_world_size())
    return 1


def _rank() -> int:
    if dist is not None and dist.is_available() and dist.is_initialized():
        return int(dist.get_rank())
    return 0


def _shard(master_weight: torch.Tensor, world: int, rank: int, dim: int) -> torch.Tensor:
    """Return this rank's contiguous partition of master_weight along ``dim``."""
    shard = master_weight.detach().chunk(world, dim=dim)[rank]
    return shard.clone().contiguous()


class _GatherLastDim(torch.autograd.Function):
    """all_gather along the last dimension.

    forward:  [x_0 | x_1 | ... | x_{W-1}]  (concatenated)
    backward: return only this rank's slice of the upstream gradient,
    which is the correct local gradient for the sharded weight / bias.
    """

    @staticmethod
    def forward(ctx, x):
        world = _world_size()
        ctx.world = world
        ctx.rank = _rank()
        if world == 1:
            return x
        x = x.contiguous()
        bufs = [torch.empty_like(x) for _ in range(world)]
        dist.all_gather(bufs, x)
        return torch.cat(bufs, dim=-1)

    @staticmethod
    def backward(ctx, grad_output):
        if ctx.world == 1:
            return grad_output
        shard = grad_output.shape[-1] // ctx.world
        return grad_output.narrow(-1, ctx.rank * shard, shard)


class _AllReduceSum(torch.autograd.Function):
    """all_reduce (sum) over the process group.

    forward:  sum of the partial tensors on every rank
    backward: identity (the full upstream gradient is the local gradient,
    because each rank's partial used the local weight shard).
    """

    @staticmethod
    def forward(ctx, x):
        ctx.world = _world_size()
        if ctx.world == 1:
            return x
        y = x.contiguous()
        dist.all_reduce(y, op=dist.ReduceOp.SUM)
        return y

    @staticmethod
    def backward(ctx, grad_output):
        return grad_output


class ColumnParallelLinear(nn.Module):
    """Linear layer with column-parallel (output-dim) weight sharding.

    y = all_gather( x @ W_shard^T + b_shard ) over ranks
    """

    def __init__(self, in_features, out_features, bias, master_weight):
        super().__init__()
        world = _world_size()
        if out_features % world != 0:
            raise ValueError(
                f"out_features ({out_features}) must be divisible by world_size ({world}) "
                "for column-parallel sharding"
            )
        rank = _rank()
        self.in_features = in_features
        self.out_features = out_features
        self.out_features_per_rank = out_features // world
        weight_shard = _shard(master_weight, world, rank, dim=0)
        self.weight = nn.Parameter(weight_shard)
        if bias:
            self.bias = nn.Parameter(
                torch.zeros(self.out_features_per_rank, dtype=weight_shard.dtype, device=weight_shard.device)
            )
        else:
            self.register_parameter("bias", None)

    def forward(self, input):
        out = F.linear(input, self.weight, self.bias)
        return _GatherLastDim.apply(out)


class RowParallelLinear(nn.Module):
    """Linear layer with row-parallel (input-dim) weight sharding.

    y = all_reduce( x_shard @ W_shard^T ) + b, with b full on every rank.

    forward accepts either the full input (..., in_features), of which
    this rank takes its partition, or an input that is already
    sharded along the last dimension (..., in_features / world_size).
    """

    def __init__(self, in_features, out_features, bias, master_weight):
        super().__init__()
        world = _world_size()
        if in_features % world != 0:
            raise ValueError(
                f"in_features ({in_features}) must be divisible by world_size ({world}) "
                "for row-parallel sharding"
            )
        rank = _rank()
        self.in_features = in_features
        self.out_features = out_features
        self.in_features_per_rank = in_features // world
        weight_shard = _shard(master_weight, world, rank, dim=1)
        self.weight = nn.Parameter(weight_shard)
        if bias:
            self.bias = nn.Parameter(
                torch.zeros(out_features, dtype=weight_shard.dtype, device=weight_shard.device)
            )
        else:
            self.register_parameter("bias", None)

    def forward(self, input):
        world = _world_size()
        last_dim = input.shape[-1]
        if last_dim == self.in_features:
            # full input: take this rank's input partition
            x = input.chunk(world, dim=-1)[_rank()]
        elif last_dim == self.in_features_per_rank:
            # already sharded input
            x = input
        else:
            raise ValueError(
                f"last dim of input ({last_dim}) must be in_features ({self.in_features}) "
                f"or in_features/world_size ({self.in_features_per_rank})"
            )
        partial = F.linear(x, self.weight)
        partial = _AllReduceSum.apply(partial)
        if self.bias is not None:
            partial = partial + self.bias
        return partial
```
