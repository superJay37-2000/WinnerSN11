"""
Attention-Based Multiple Instance Learning (ABMIL) Implementation

This script contains the complete implementation of ABMIL for computational pathology
.
Reference: Ilse, M., Tomczak, J., & Welling, M. (2018). Attention-based deep
multiple instance learning. In ICML (pp. 2127-2136). PMLR.
"""

import torch
import torch.nn as nn
import torch.nn.functional as F


_FLOAT_DTYPES = (torch.float16, torch.bfloat16, torch.float32, torch.float64)


def _compute_dtype(dtype):
    """Pick a numerically-safe working precision for attention arithmetic.

    Low-precision floating point is promoted so that the softmax of the
    attention scores (and the weighted average) is accurate to the tolerance
    appropriate for the *output* dtype (roughly 1e-2 / 1e-6 / 1e-10 for
    float16 / float32 / float64).
    """
    if dtype in (torch.float32, torch.float64):
        return dtype
    return torch.float32  # float16 / bfloat16 -> float32


class Attention_TanhSigmoidGating(nn.Module):
    def __init__(self, D=64, L=64, dropout=0.25):
        r"""
        Global attention pooling layer with tanh non-linearity and sigmoid gating
        (Ilse et al. 2018).

        Args:
            D (int): input feature dimension.
            L (int): hidden layer dimension. Notation changed from M from Ilse et
 al
                2018, as M is overloaded to also describe # of patch embeddings in W
SI.
            dropout (float): Dropout probability.

        Returns:
            A_norm (torch.Tensor): [M x 1]-dim tensor of normalized attention scores
                (sum to 1)
        """
        super(Attention_TanhSigmoidGating, self).__init__()
        if D <= 0 or L <= 0:
            raise ValueError("D and L must be positive integers")
        self.D = D
        self.L = L
        self.tanhV = nn.Sequential(
            nn.Linear(D, L),
            nn.Tanh(),
            nn.Dropout(dropout),
        )
        self.sigmU = nn.Sequential(
            nn.Linear(D, L),
            nn.Sigmoid(),
            nn.Dropout(dropout),
        )
        self.w = nn.Linear(L, 1)

    def forward(self, H):
        r"""
        Compute normalized attention scores for a bag of instances.

        Args:
            H (torch.Tensor): [M x D]-dim tensor of instance features.

        Returns:
            A_norm (torch.Tensor): [M x 1]-dim tensor of normalized attention
                scores (softmax, sums to 1).
        """
        if not torch.is_tensor(H):
            raise TypeError("H must be a torch.Tensor")
        if H.dim() != 2:
            raise ValueError(
                "H must be a 2-D tensor of shape [M x D], got shape {}".format(tuple(H.shape
))
            )
        M, D_in = H.shape
        if D_in != self.D:
            raise ValueError(
                "H has feature dimension {}, expected {} (self.D)".format(D_in, self
.D)
            )
        if M < 1:
            raise ValueError("A bag must contain at least one instance, got M=0")
        if H.dtype not in _FLOAT_DTYPES:
            raise TypeError("H must be a floating-point tensor, got dtype {}".format(H
.dtype))
        if not bool(torch.isfinite(H).all()):
            raise ValueError("H contains NaN or Inf values")

        compute_dtype = _compute_dtype(H.dtype)
        if compute_dtype == H.dtype:
            tanhV, sigmU, w = self.tanhV, self.sigmU, self.w
            H_c = H
        else:
            tanhV = self.tanhV.to(compute_dtype)
            sigmU = self.sigmU.to(compute_dtype)
            w = self.w.to(compute_dtype)
            H_c = H.to(compute_dtype)

        tanh_h = tanhV(H_c)    # [M x L]
        sigm_h = sigmU(H_c)    # [M x L]
        a = w(tanh_h * sigm_h)  # [M x 1]  (a_i = w^T (tanh(V h_i) ⊙ sigmoid(U h_i)))
        A_raw = a.squeeze(-1)   # [M]
        # softmax is numerically stable (max-subtraction is done internally)
        A_norm = F.softmax(A_raw, dim=0)  # [M]
        return A_norm.to(H.dtype).view(-1, 1)

# -*- coding: utf-8 -*-
"""
Attention-Based Multiple Instance Learning (ABMIL) Implementation

This script contains the complete implementation of ABMIL for computational pathology.
Reference: Ilse, M., Tomczak, J., & Welling, M. (2018). Attention-based deep
multiple instance learning. In ICML (pp. 2127-2136). PMLR.
"""

import torch
import torch.nn as nn
import torch.nn.functional as F


_FLOAT_DTYPES = (torch.float16, torch.bfloat16, torch.float32, torch.float64)


def _compute_dtype(dtype):
    """Pick a numerically-safe working precision for attention arithmetic.

    Low-precision floating point is promoted so that the softmax of the
    attention scores (and the weighted average) is accurate to the tolerance
    appropriate for the *output* dtype (roughly 1e-2 / 1e-6 / 1e-10 for
    float16 / float32 / float64).
    """
    if dtype in (torch.float32, torch.float64):
        return dtype
    return torch.float32  # float16 / bfloat16 -> float32


class Attention_TanhSigmoidGating(nn.Module):
    def __init__(self, D=64, L=64, dropout=0.25):
        r"""
        Global attention pooling layer with tanh non-linearity and sigmoid gating
        (Ilse et al. 2018).

        Args:
            D (int): input feature dimension.
            L (int): hidden layer dimension. Notation changed from M from Ilse et al
                2018, as M is overloaded to also describe # of patch embeddings in WSI.
            dropout (float): Dropout probability.

        Returns:
            A_norm (torch.Tensor): [M x 1]-dim tensor of normalized attention scores
                (sum to 1)
        """
        super(Attention_TanhSigmoidGating, self).__init__()
        if D <= 0 or L <= 0:
            raise ValueError("D and L must be positive integers")
        self.D = D
        self.L = L
        self.tanhV = nn.Sequential(
            nn.Linear(D, L),
            nn.Tanh(),
            nn.Dropout(dropout),
        )
        self.sigmU = nn.Sequential(
            nn.Linear(D, L),
            nn.Sigmoid(),
            nn.Dropout(dropout),
        )
        self.w = nn.Linear(L, 1)

    def forward(self, H):
        r"""
        Compute normalized attention scores for a bag of instances.

        Args:
            H (torch.Tensor): [M x D]-dim tensor of instance features.

        Returns:
            A_norm (torch.Tensor): [M x 1]-dim tensor of normalized attention
                scores (softmax, sums to 1).
        """
        if not torch.is_tensor(H):
            raise TypeError("H must be a torch.Tensor")
        if H.dim() != 2:
            raise ValueError(
                "H must be a 2-D tensor of shape [M x D], got shape {}".format(tuple(H.shape))
            )
        M, D_in = H.shape
        if D_in != self.D:
            raise ValueError(
                "H has feature dimension {}, expected {} (self.D)".format(D_in, self.D)
            )
        if M < 1:
            raise ValueError("A bag must contain at least one instance, got M=0")
        if H.dtype not in _FLOAT_DTYPES:
            raise TypeError("H must be a floating-point tensor, got dtype {}".format(H.dtype))
        if not bool(torch.isfinite(H).all()):
            raise ValueError("H contains NaN or Inf values")

        compute_dtype = _compute_dtype(H.dtype)
        if compute_dtype == H.dtype:
            tanhV, sigmU, w = self.tanhV, self.sigmU, self.w
            H_c = H
        else:
            tanhV = self.tanhV.to(compute_dtype)
            sigmU = self.sigmU.to(compute_dtype)
            w = self.w.to(compute_dtype)
            H_c = H.to(compute_dtype)

        tanh_h = tanhV(H_c)    # [M x L]
        sigm_h = sigmU(H_c)    # [M x L]
        a = w(tanh_h * sigm_h)  # [M x 1]  (a_i = w^T (tanh(V h_i) ⊙ sigmoid(U h_i)))
        A_raw = a.squeeze(-1)   # [M]
        # softmax is numerically stable (max-subtraction is done internally)
        A_norm = F.softmax(A_raw, dim=0)  # [M]
        return A_norm.to(H.dtype).view(-1, 1)
